    public static void abrirConexao() {
        if (!conexaoAberta) {
            int retorno = ImpressoraDLL.INSTANCE.AbreConexaoImpressora(1, "i9", "USB", 0);

            }
        } else {

        }
    }

    public static void fecharConexao() {
        if (conexaoAberta) {
